    return 0;   
}

## การคำนวณทางคณิตศาสตร์
### ภาษา C
```c
#include <stdio.h>
int main(void){
    // กำหนดค่าตัวแปร a, b และ c
    int a=65,b=20,c;
    // คำนวณค่า c = a + b
    c=a+b;
    // แสดงผลลัพธ์ทางจอภาพ
    printf("a = %d\nb = %d\nc = a + b = %d\n", a, b, c);
    return 0;
}

```

ให้นักศึกษาเขียนโปรแกรมเพื่อคำนวณ ดังนี้
1. ผลรวมของของชุด: 10,20,30,40,50
2. หาค่าเฉลี่ย ของชุดข้อมูล: 22,33,44,55,66
3. หาค่าเป็นเปอร์เซนต์ ชุดข้อมูล 30, 10, 40, 50, 10, 9
